# 录像存储与时间轴回放

> **适合读者**：需要实现视频录制、存储和历史回放的后端开发者  
> **核心功能**：连续录像 + 按时间段检索 + 时间轴拖拽回放

---

## 一、录像系统核心设计

```
摄像头 RTSP
    │
    ▼
[录像进程]  ──→  [存储层]   ──→  [索引层]
FFmpeg 按时间   NFS/OSS/本地    PostgreSQL
分段录制        TS 切片文件      记录每个片段的
                                时间范围
    │
    ▼
[回放接口]
按时间查询片段 → 拼接 M3U8 → 前端时间轴播放
```

## 二、分段录像（FFmpeg 实现）

```bash
# 每 10 分钟切一个文件，按时间命名
ffmpeg -rtsp_transport tcp \
  -i "rtsp://admin:pass@192.168.1.100/stream" \
  -c copy \
  -f segment \
  -segment_time 600 \                              # 每段 600 秒 = 10 分钟
  -segment_format mp4 \
  -movflags frag_keyframe+empty_moov \             # fMP4：moov 前置，掉电/杀进程不至整段损坏
  -segment_atclocktime 1 \                         # 整点对齐（如 14:00, 14:10）
  -reset_timestamps 1 \
  -strftime 1 \
  "/recordings/camera01/%Y%m%d_%H%M%S.mp4"        # 按时间命名（分段式 MP4）
```

生成文件示例：
```
/recordings/camera01/
  20260729_140000.mp4   (14:00-14:10)
  20260729_141000.mp4   (14:10-14:20)
  20260729_142000.mp4   (14:20-14:30)
```

## 三、索引入库（Go 实现）

```go
// 录像文件元数据
type RecordingSegment struct {
    SegmentID  string    `gorm:"column:segment_id;primaryKey"`
    CameraID   string    `gorm:"column:camera_id"`
    FilePath   string    `gorm:"column:file_path"`
    StartTime  int64     `gorm:"column:start_time"`  // 毫秒时间戳
    EndTime    int64     `gorm:"column:end_time"`    // 毫秒时间戳
    Duration   int       `gorm:"column:duration"`    // 秒
    FileSize   int64     `gorm:"column:file_size"`   // 字节
    CreatedAt  int64     `gorm:"column:created_at;autoCreateTime:milli"`
}

// 监控录像目录，新文件完成后入库
func (s *RecordingService) WatchRecordingDir(ctx context.Context, cameraID, dir string) {
    watcher, _ := fsnotify.NewWatcher()
    watcher.Add(dir)

    for {
        select {
        case event := <-watcher.Events:
            if event.Op == fsnotify.Create && strings.HasSuffix(event.Name, ".mp4") {
                // 等待文件写完（FFmpeg 切换到下一段时上一段才完整）
                time.Sleep(2 * time.Second)
                s.indexSegment(ctx, cameraID, event.Name)
            }
        case <-ctx.Done():
            watcher.Close()
            return
        }
    }
}

func (s *RecordingService) indexSegment(ctx context.Context, cameraID, filePath string) {
    // 用 FFprobe 获取视频时长
    out, err := exec.Command("ffprobe",
        "-v", "quiet", "-print_format", "json",
        "-show_format", filePath,
    ).Output()
    if err != nil {
        log.Error().Err(err).Str("path", filePath).Msg("ffprobe 失败")
        return
    }

    var probe struct {
        Format struct {
            Duration string `json:"duration"`
            Size     string `json:"size"`
        } `json:"format"`
    }
    json.Unmarshal(out, &probe)

    // 从文件名解析开始时间
    basename := filepath.Base(filePath)                        // 20260729_140000.mp4
    t, _ := time.ParseInLocation("20060102_150405", basename[:15], time.Local)

    dur, _ := strconv.ParseFloat(probe.Format.Duration, 64)
    size, _ := strconv.ParseInt(probe.Format.Size, 10, 64)

    seg := &model.RecordingSegment{
        SegmentID: idgen.GenerateUUID(),
        CameraID:  cameraID,
        FilePath:  filePath,
        StartTime: t.UnixMilli(),
        EndTime:   t.Add(time.Duration(dur * float64(time.Second))).UnixMilli(),
        Duration:  int(dur),
        FileSize:  size,
    }

    if err := s.repo.Create(ctx, seg); err != nil {
        log.Error().Err(err).Str("path", filePath).Msg("录像索引入库失败")
    }
}
```

## 四、时间轴回放接口

```go
// 查询指定时间段的录像片段，生成回放 M3U8
func (s *RecordingService) GetPlaybackM3U8(ctx context.Context, cameraID string, startMS, endMS int64) (string, error) {
    // 查询覆盖目标时间段的所有片段
    segments, err := s.repo.GetSegmentsByTimeRange(ctx, cameraID, startMS, endMS)
    if err != nil {
        return "", xerr.Wrap(xerr.ErrInternal.Code, "查询录像失败", err)
    }
    if len(segments) == 0 {
        return "", xerr.New(xerr.ErrNotFound.Code, "该时间段无录像")
    }

    // 生成 VOD M3U8
    var sb strings.Builder
    sb.WriteString("#EXTM3U\n")
    sb.WriteString("#EXT-X-VERSION:3\n")
    sb.WriteString("#EXT-X-TARGETDURATION:600\n")
    sb.WriteString("#EXT-X-PLAYLIST-TYPE:VOD\n")

    for _, seg := range segments {
        // 生成带鉴权的文件 URL
        url := s.generateFileURL(seg.FilePath, seg.StartTime, seg.EndTime)
        sb.WriteString(fmt.Sprintf("#EXTINF:%.1f,\n%s\n", float64(seg.Duration), url))
    }

    sb.WriteString("#EXT-X-ENDLIST\n")
    return sb.String(), nil
}
```

**Repository 查询**：

```go
func (r *RecordingRepo) GetSegmentsByTimeRange(ctx context.Context, cameraID string, startMS, endMS int64) ([]*model.RecordingSegment, error) {
    return r.q.RecordingSegment.WithContext(ctx).
        Where(r.q.RecordingSegment.CameraID.Eq(cameraID)).
        // 片段与查询时间段有交集：片段开始 < 查询结束 AND 片段结束 > 查询开始
        Where(r.q.RecordingSegment.StartTime.Lt(endMS)).
        Where(r.q.RecordingSegment.EndTime.Gt(startMS)).
        Order(r.q.RecordingSegment.StartTime.Asc()).
        Find()
}
```

## 五、存储管理（自动清理）

```go
// 清理超过保留天数的录像（每天凌晨执行）
func (s *RecordingService) CleanExpiredRecordings(ctx context.Context, retentionDays int) error {
    cutoff := time.Now().AddDate(0, 0, -retentionDays).UnixMilli()

    // 查询过期文件
    segments, err := s.repo.GetSegmentsBefore(ctx, cutoff)
    if err != nil {
        return xerr.Wrap(xerr.ErrInternal.Code, "查询过期录像失败", err)
    }

    for _, seg := range segments {
        // 删除文件
        if err := os.Remove(seg.FilePath); err != nil && !os.IsNotExist(err) {
            log.Warn().Err(err).Str("path", seg.FilePath).Msg("删除录像文件失败")
            continue
        }
        // 删除数据库记录
        if err := s.repo.Delete(ctx, seg.SegmentID); err != nil {
            log.Error().Err(err).Str("id", seg.SegmentID).Msg("删除录像记录失败")
        }
    }

    log.Info().Int("count", len(segments)).Msg("清理过期录像完成")
    return nil
}
```

## 六、前端时间轴组件

```javascript
// 简单时间轴：点击某时刻跳转回放
class TimelinePlayer {
  constructor(videoEl, cameraID) {
    this.video = videoEl;
    this.cameraID = cameraID;
  }

  // 跳转到指定时间点
  async seek(timestamp) {
    const WIN_MS = 5 * 60 * 1000;              // 窗口起点向前开 5 分钟（起播前的缓冲）
    const startMS = timestamp - WIN_MS;        // playlist 起点
    const endMS = timestamp + 60 * 60 * 1000;  // 窗口终点：目标时刻后 1 小时

    const resp = await fetch(
      `/api/v1/recordings/playback?camera_id=${this.cameraID}&start=${startMS}&end=${endMS}`
    );
    const { data } = await resp.json();
    const m3u8URL = data.m3u8_url;

    // 用 hls.js 播放
    const hls = new Hls();
    hls.loadSource(m3u8URL);
    hls.attachMedia(this.video);
    hls.on(Hls.Events.MANIFEST_PARSED, () => {
      // 播放器时间轴以 playlist 第一个片段为 0 点，
      // 目标时刻偏移 = 目标时刻 - 窗口起点（此处恒为 WIN_MS/1000 = 300s，正值目标时刻）
      const offsetSeconds = (timestamp - startMS) / 1000;
      this.video.currentTime = offsetSeconds;
      this.video.play();
    });
  }
}
```

## 七、录像回放要懂的三件事

### 1. 容器：为什么录像不能用裸 MP4

- MP4 的索引（moov）写在文件**末尾**。掉电、被 kill、推流中断都会导致文件没写完 moov → 整段不可播放——排查表里"文件末尾损坏"的根因就是它。
- 录像领域普遍用 **MPEG-TS**（.ts）：按数据流顺序排列，任何干净的切片点都能续读，且本身就是 HLS 的分片格式，录完可直接保存供播放。
- 必须用 MP4 时（体积小、便于点播复用），用**分段式 MP4（fMP4）**：`-movflags frag_keyframe+empty_moov` 把片段元数据前置，损坏范围从"整段"缩小到"断点所在的一个分片"。第二节命令已加该参数。

### 2. 两种录像：连续 vs 事件触发

| 类型 | 触发 | 覆盖范围 | 存储形态 |
|------|------|---------|---------|
| 连续录像 | 恒定拉流不间断 | 全天 24h | 本文主线：固定时长切片 |
| 事件/告警录像 | 运动检测、AI 报警 | 事件前 N 秒 ~ 事件后 N 秒 | 短切片 + 事件元数据 |

- 事件录像的回放入口是"事件列表"而非纯时间轴；索引表加一列 `event_type` 即可区分。
- 常见做法是两种都录：连续录保证据完整，事件录便于快速检索定位。

### 3. 回放不只是跳转：倍速 / 暂停 / 逐帧

- 时间轴之外，回放的核心操作是**倍速**（2x / 4x / 16x）、**暂停**、逐秒/逐帧定位。
- 概念：正向快进 = 跳过非关键帧、只播 I 帧；回退/慢放要保证入口落在关键帧上，否则要等最近的 I 帧到位才能起播。
- 前端用 hls.js 的 `video.playbackRate` 实现倍速；接入国标（GB28181）时对应 INVITE 建会话后的 PLAY / FASTFORWARD 控制指令（第 25 篇展开）。

## 八、可靠性：断流续录与文件对账

### 1. 断流续录：录像进程要有守护

- FFmpeg 拉 RTSP 是**单次进程**：摄像头断流、网络抖动、设备重启都会让进程以非 0 码退出，录像随之停止——不守护的话"连续录像"随时断在路上的某一段。
- 守护两件套：外层用 `supervisor` / `systemd` / 自研 watchdog 拉起并自动重启；FFmpeg 自身加自动重连（`-reconnect 1 -reconnect_streamed 1 -reconnect_delay_max 5`）扛住短暂抖动。
- 用流媒体服务器内置录像（SRS / MediaMTX / ZLMediaKit）时，断流自动续录由服务端负责，业务侧只关心索引；断流期间即空洞，恢复后从新时间点切片，最后由 DB 时间范围对外显示"无录像"。

### 2. 存储一致性：文件与数据库要对账

- 第五节清理流程是"先删文件、再删 DB 记录"，两步不原子：文件删了但 DB 记录删除失败 → 索引指向不存在的文件（回放 404）；文件写完但索引没入库 → 孤立文件白占磁盘。
- 对账 job（每日轮询）：以录像目录或 DB 一方为权威源，扫出另一边的差值并补平——DB 有记录而文件缺失 → 删记录；文件存在而 DB 无记录 → 补索引或清掉孤儿文件。
- 本质：**文件系统与数据库是两套状态**，所有"动文件必动 DB"的操作都不保证原子，最终一致性靠对账兜底。

## 本篇小结

| 功能 | 技术方案 |
|------|---------|
| 分段录像 | FFmpeg `-f segment -segment_time` |
| 录像索引 | PostgreSQL 存时间范围，支持范围查询 |
| 时间轴回放 | 动态拼接 VOD M3U8 |
| 存储清理 | 定时任务按保留天数清理 |
| 断流续录 | supervisor/systemd 守护 + FFmpeg 自动重连 |
| 文件/DB 一致性 | 定时对账 job，以目录或 DB 为权威源补差值 |
| 存储选型 | 内网用本地/NFS，云端用 OSS/S3 |

### 练习2：Go 写入录像索引到数据库
```go
// 录像完成回调：写入片段信息（start_time/end_time/object_key）
type RecordSegment struct {
    SegmentID  string `gorm:"column:segment_id;primaryKey"`
    CameraID   string `gorm:"column:camera_id"`
    StartTime  int64  `gorm:"column:start_time"`  // 毫秒时间戳
    EndTime    int64  `gorm:"column:end_time"`
    ObjectKey  string `gorm:"column:object_key"`  // S3/OSS路径
    Duration   int    `gorm:"column:duration"`    // 秒
}

// 查询某时间段内的所有录像片段
func (r *RecordRepo) ListByTimeRange(ctx context.Context, cameraID string, start, end int64) ([]*RecordSegment, error) {
    return r.q.RecordSegment.WithContext(ctx).
        Where(r.q.RecordSegment.CameraID.Eq(cameraID)).
        Where(r.q.RecordSegment.StartTime.Lte(end)).
        Where(r.q.RecordSegment.EndTime.Gte(start)).
        Order(r.q.RecordSegment.StartTime.Asc()).
        Find()
}
```
**预期输出**：查询返回所有与时间窗口有交集的录像片段，按开始时间升序排列。

### 练习3：动态拼接 VOD M3U8（时间轴回放）
```go
// 将多个 MP4 片段拼接为一个 VOD m3u8 供前端回放
func BuildVODM3U8(segments []*RecordSegment, baseURL string) string {
    var sb strings.Builder
    sb.WriteString("#EXTM3U
#EXT-X-VERSION:3
#EXT-X-TARGETDURATION:300
")
    for _, seg := range segments {
        durationSec := float64(seg.Duration)
        sb.WriteString(fmt.Sprintf("#EXTINF:%.3f,
%s/%s
", durationSec, baseURL, seg.ObjectKey))
    }
    sb.WriteString("#EXT-X-ENDLIST
")
    return sb.String()
}
```
**预期输出**：返回标准 HLS m3u8 内容，播放器可直接播放指定时间范围的历史录像。

## 常见问题排查

| 现象 | 可能原因 | 解决方法 |
|------|---------|---------|
| 录像文件末尾损坏 | FFmpeg 被强制 kill（未 Ctrl+C）| 用 `kill -INT` 发送 SIGINT，FFmpeg 会写入文件尾 |
| 时间段内有录像空白 | 摄像头推流中断未被检测 | 数据库中片段的时间范围对比，前端在空白处显示"无录像" |
| 磁盘满导致录像停止 | 未配置自动清理策略 | 定时任务按保留天数删除旧片段（同步更新数据库记录）|
| VOD m3u8 播放跳帧 | 片段拼接处没有关键帧 | 录像时强制按 GOP 边界切片（`-segment_atclocktime 1` + `-force_key_frames "expr:gte(t,0)"`）|
| S3/OSS 直链403 | Bucket 不允许公开访问 | 使用预签名 URL（Presigned URL），设置5-60分钟有效期 |
