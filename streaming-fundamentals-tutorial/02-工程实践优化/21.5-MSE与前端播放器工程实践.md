## 一、MSE API 深入

### 1.1 为什么需要 MSE

浏览器原生 `<video>` 标签只支持完整文件（通过 HTTP Range 请求）和几种固定协议（HLS/DASH）。**RTSP/RTMP/FLV 流无法直接播放**，因为：

```
浏览器 <video src="rtsp://camera/stream">  ❌ 不支持
浏览器 <video src="stream.flv">            ❌ 需要完整文件
```

**MSE（Media Source Extensions）** 允许 JavaScript 动态喂数据给 video 元素：

```
WebSocket/HTTP 接收 FLV 流 → JS 解复用 → MSE appendBuffer → video 播放
```

### 1.2 MSE 核心 API

```javascript
// 1. 创建 MediaSource 对象
const mediaSource = new MediaSource();
const video = document.getElementById('video');
video.src = URL.createObjectURL(mediaSource);

// 2. 等待 sourceopen 事件（MediaSource 就绪）
mediaSource.addEventListener('sourceopen', () => {
  // 3. 添加 SourceBuffer（指定 codec）
  const sourceBuffer = mediaSource.addSourceBuffer(
    'video/mp4; codecs="avc1.42E01E, mp4a.40.2"'  // H.264 + AAC
  );

  // 4. 监听 updateend（appendBuffer 完成后才能再次 append）
  sourceBuffer.addEventListener('updateend', () => {
    if (hasMoreData && !sourceBuffer.updating) {
      sourceBuffer.appendBuffer(nextChunk);  // 继续喂数据
    }
  });

  // 5. 喂数据（必须是 Uint8Array）
  sourceBuffer.appendBuffer(uint8ArrayData);
});
```

**状态机**：

```
MediaSource 状态：
  closed → open（可以 addSourceBuffer）→ ended（播放完成）

SourceBuffer 状态：
  updating = false → appendBuffer() → updating = true → updateend 事件 → updating = false
```

### 1.3 浏览器兼容性检测

```javascript
// 检测 MSE 支持
if (!window.MediaSource) {
  console.error('MSE not supported');
}

// 检测特定 codec 支持
const codec = 'video/mp4; codecs="avc1.42E01E, mp4a.40.2"';
if (!MediaSource.isTypeSupported(codec)) {
  console.error('H.264 + AAC not supported');
}
```

## 二、mpegts.js 原理拆解

### 2.1 mpegts.js 的工作流程

```
1. WebSocket 接收 FLV 二进制流
   ↓
2. FLV 解复用（Demuxer）
   - 解析 FLV Tag（VideoTag/AudioTag）
   - 提取 H.264 NALU + AAC 帧
   ↓
3. 转换为 ISO BMFF（fMP4）片段
   - 生成 ftyp + moov（初始化段）
   - 生成 moof + mdat（媒体片段）
   ↓
4. appendBuffer 到 SourceBuffer
   ↓
5. video 元素播放
```

**关键点**：FLV 本身不能直接 append 到 SourceBuffer，必须转换为 fMP4 或 MPEG-2 TS。

### 2.2 为什么不能直接 append FLV

```javascript
// ❌ 错误：SourceBuffer 不支持 FLV MIME 类型
sourceBuffer.addSourceBuffer('video/x-flv');  // 抛异常

// ✅ 正确：必须转换为 fMP4
sourceBuffer.addSourceBuffer('video/mp4; codecs="avc1.42E01E, mp4a.40.2"');
```

### 2.3 mpegts.js 的配置选项

```javascript
const player = mpegts.createPlayer({
  type: 'flv',           // 输入格式：flv/m2ts
  isLive: true,          // 直播流
  url: 'ws://server/stream.flv',
}, {
  // 解复用配置
  enableWorker: true,                     // 在 Worker 线程解复用（推荐）
  enableStashBuffer: false,               // 直播关闭预读缓冲

  // 低延迟配置
  liveBufferLatencyChasing: true,         // 开启追帧
  liveBufferLatencyMaxLatency: 1.5,       // 最大延迟 1.5s
  liveBufferLatencyMinRemain: 0.5,        // 最小保留 0.5s

  // 内存管理
  autoCleanupSourceBuffer: true,          // 自动清理旧数据
  autoCleanupMaxBackwardDuration: 30,     // 保留最近 30 秒
  autoCleanupMinBackwardDuration: 20,     // 缓冲小于 20 秒时才清理
});
```

## 三、完整的播放器封装

### 3.1 生产级 StreamPlayer 类

```javascript
class StreamPlayer {
  constructor(video, url, options = {}) {
    this.video = video;
    this.url = url;
    this.player = null;
    this.reconnectAttempts = 0;
    this.maxReconnectAttempts = options.maxReconnectAttempts || 5;
    this.destroyed = false;
  }

  init() {
    // MSE 兼容性检测
    if (!mpegts.isSupported()) {
      console.error('MSE not supported in this browser');
      throw new Error('MSE not supported');
    }

    this.player = mpegts.createPlayer({
      type: 'flv',
      isLive: true,
      url: this.url,
    }, {
      enableWorker: true,
      enableStashBuffer: false,
      liveBufferLatencyChasing: true,
      liveBufferLatencyMaxLatency: 1.5,
      liveBufferLatencyMinRemain: 0.5,
    });

    // 绑定事件
    this.player.on(mpegts.Events.ERROR, this.handleError.bind(this));
    this.player.on(mpegts.Events.LOADING_COMPLETE, () => {
      console.log('Stream loading complete');
    });
    this.player.on(mpegts.Events.STATISTICS_INFO, (stats) => {
      // 监控统计信息（可选）
      console.debug('Buffer:', stats.currentSegmentIndex, 'Speed:', stats.speed);
    });

    this.player.attachMediaElement(this.video);
    this.player.load();
    this.player.play().catch(err => {
      console.warn('Autoplay blocked:', err);
      // 浏览器自动播放策略阻止，需要用户交互
    });

    this.reconnectAttempts = 0;  // 重置重连计数
  }

  handleError(errType, errDetail) {
    console.error('Player error:', errType, errDetail);

    if (errDetail === mpegts.ErrorDetails.NETWORK_EXCEPTION) {
      this.reconnect();
    } else if (errDetail === mpegts.ErrorDetails.MEDIA_MSE_ERROR) {
      console.error('MSE error, may need to refresh page');
      // MSE 错误通常需要重新创建 MediaSource
      this.destroy();
      setTimeout(() => this.init(), 1000);
    }
  }

  reconnect() {
    if (this.destroyed) return;

    if (this.reconnectAttempts >= this.maxReconnectAttempts) {
      console.error('Max reconnect attempts reached');
      return;
    }

    this.destroy();

    // 指数退避
    const delay = Math.min(1000 * Math.pow(2, this.reconnectAttempts), 30000);
    console.log(`Reconnecting in ${delay}ms (attempt ${this.reconnectAttempts + 1})`);

    setTimeout(() => {
      this.reconnectAttempts++;
      this.init();
    }, delay);
  }

  destroy() {
    if (this.player) {
      try {
        this.player.pause();
        this.player.unload();
        this.player.detachMediaElement();
        this.player.destroy();
      } catch (e) {
        console.warn('Error destroying player:', e);
      }
      this.player = null;
    }
  }

  dispose() {
    this.destroyed = true;
    this.destroy();
  }
}

// 使用示例
const player = new StreamPlayer(
  document.getElementById('video'),
  'ws://localhost:8080/live/stream.flv'
);
player.init();

// 切换流时必须先销毁
player.destroy();
```

### 3.2 为什么必须完整销毁

```javascript
// ❌ 错误：只调用 pause，不销毁
player.pause();
newPlayer = mpegts.createPlayer(...);  // 旧的 SourceBuffer 仍占用内存

// ✅ 正确：完整销毁流程
player.pause();          // 1. 停止播放
player.unload();         // 2. 清空 SourceBuffer
player.detachMediaElement();  // 3. 解绑 video 元素
player.destroy();        // 4. 释放所有资源
player = null;           // 5. 清空引用
```

## 四、16 路流管理

### 4.1 MultiStreamManager 类

```javascript
class MultiStreamManager {
  constructor() {
    this.players = new Map();  // streamID → StreamPlayer
  }

  addStream(streamID, videoElement, url) {
    // 如果已存在，先销毁旧的
    if (this.players.has(streamID)) {
      this.removeStream(streamID);
    }

    const player = new StreamPlayer(videoElement, url, {
      maxReconnectAttempts: 3,  // 多路流重连次数可以少一些
    });

    try {
      player.init();
      this.players.set(streamID, player);
    } catch (err) {
      console.error(`Failed to init stream ${streamID}:`, err);
    }
  }

  removeStream(streamID) {
    const player = this.players.get(streamID);
    if (player) {
      player.dispose();
      this.players.delete(streamID);
    }
  }

  // 按需激活：只播放可见的流
  pauseInvisibleStreams(visibleStreamIDs) {
    for (const [streamID, player] of this.players.entries()) {
      if (!visibleStreamIDs.includes(streamID) && player.player) {
        player.player.pause();
      }
    }
  }

  resumeVisibleStreams(visibleStreamIDs) {
    for (const streamID of visibleStreamIDs) {
      const player = this.players.get(streamID);
      if (player && player.player) {
        player.player.play().catch(console.warn);
      }
    }
  }

  destroyAll() {
    for (const player of this.players.values()) {
      player.dispose();
    }
    this.players.clear();
  }
}

// 16 路分屏使用
const manager = new MultiStreamManager();

// 添加 16 路流
for (let i = 1; i <= 16; i++) {
  const video = document.getElementById(`video-${i}`);
  manager.addStream(`cam${i}`, video, `ws://server/live/cam${i}.flv`);
}

// 切换分屏布局时（如 16 → 9），只保留可见的
manager.pauseInvisibleStreams(['cam1', 'cam2', 'cam3', 'cam4', 'cam5', 'cam6', 'cam7', 'cam8', 'cam9']);

// 页面卸载时销毁
window.addEventListener('beforeunload', () => {
  manager.destroyAll();
});
```

### 4.2 按需激活策略

```javascript
// 场景：4/9/16 分屏切换
const layouts = {
  '4': ['cam1', 'cam2', 'cam3', 'cam4'],
  '9': ['cam1', 'cam2', 'cam3', 'cam4', 'cam5', 'cam6', 'cam7', 'cam8', 'cam9'],
  '16': [...Array(16)].map((_, i) => `cam${i + 1}`),
};

function switchLayout(layoutType) {
  const visibleStreams = layouts[layoutType];

  // 暂停不可见的流
  manager.pauseInvisibleStreams(visibleStreams);

  // 恢复可见的流
  manager.resumeVisibleStreams(visibleStreams);

  // 更新 UI 布局
  document.body.className = `layout-${layoutType}`;
}
```

## 五、MSE 的限制与对策

| 限制 | 现象 | 解决方案 |
|------|------|----------|
| **Codec 必须浏览器支持** | 某些 H.265/VP9 不播放 | 服务端统一转码为 H.264 + AAC |
| **SourceBuffer 数量限制** | 16 路后新流打不开 | 按需激活，不可见流调用 `player.destroy()` |
| **内存持续增长** | 长时间播放内存爆 | 定期清理旧数据：`sourceBuffer.remove(0, currentTime - 30)` |
| **QuotaExceededError** | appendBuffer 时抛异常 | 检测 `sourceBuffer.buffered.length`，超过阈值先 remove |
| **Safari 的 CORS 限制** | 跨域 WebSocket 不工作 | 服务端配置 CORS 头 |

### 5.1 内存清理配置

```javascript
// mpegts.js 内部会自动清理，但可以手动配置
const player = mpegts.createPlayer(config, {
  autoCleanupSourceBuffer: true,      // 自动清理

  autoCleanupMinBackwardDuration: 20,  // 缓冲小于 20 秒时才清理
});
```

## 六、MSE 硬解码限制与 WASM 回退机制

### 6.1 MSE 硬解码的并发数限制

**关键限制**：浏览器对 MSE 硬解码的视频实例数有隐性限制，超过后无法创建新的播放器。

**实测数据（Chrome 125，Intel i7-12700）**：

| 浏览器 | 操作系统 | MSE 硬解并发上限 | 超限后现象 |
|--------|---------|-----------------|-----------|
| **Chrome 125** | Windows 11 | ~8 个 video 实例 | 第 9 个黑屏或报错 |
| **Chrome 125** | macOS 14 | ~6 个 video 实例 | 第 7 个无法播放 |
| **Firefox 130** | Windows 11 | ~10 个 video 实例 | 第 11 个卡顿 |
| **Safari 17** | macOS 14 | ~12 个 video 实例 | 第 13 个降级软解 |

**原因**：
- 硬解码依赖 GPU 硬件加速，GPU 解码器数量有限
- 浏览器内部对 MediaSource 数量有保护性限制
- 不同显卡的硬件解码器数量不同（集显 < 独显）

**验证方法**：

```javascript
// 创建多个 mpegts.js 播放器，观察第 N 个是否能播放
const players = [];
for (let i = 1; i <= 16; i++) {
  const video = document.getElementById(`video-${i}`);
  const player = mpegts.createPlayer({
    type: 'flv',
    isLive: true,
    url: `http://server/live/cam${i}.flv`,
  }, {
    enableWorker: true,
    enableStashBuffer: false,
  });
  
  player.on(mpegts.Events.ERROR, (errorType, errorDetail) => {
    console.error(`Player ${i} error:`, errorType, errorDetail);
  });
  
  player.attachMediaElement(video);
  player.load();
  players.push(player);
}

// 观察 Chrome DevTools Console，看第几个开始报错
// 典型错误：QuotaExceededError, MediaError, SourceBuffer error
```

### 6.2 H.265 在 MSE 中的支持现状

**核心问题**：MSE API 本身不限制编码格式，但浏览器对 H.265 支持极其有限。

**浏览器兼容性（2026 年现状）**：

| 浏览器 | H.264 (AVC) | H.265 (HEVC) | 检测方法 |
|--------|-------------|--------------|---------|
| **Chrome 125+** | ✅ 完全支持 | ❌ 完全不支持 | `isTypeSupported('video/mp4; codecs="hev1.1.6.L93.B0"')` → false |
| **Firefox 130+** | ✅ 完全支持 | ❌ 完全不支持 | 同上 |
| **Safari 17+ (macOS)** | ✅ 完全支持 | ⚠️ **部分支持** | `hvc1` 支持，`hev1` 不支持 |
| **Safari 17+ (iOS)** | ✅ 完全支持 | ✅ **支持** | 硬件解码 H.265 |
| **Edge 125+** | ✅ 完全支持 | ❌ 完全不支持 | 同 Chrome（基于 Chromium） |

**H.265 Codec 字符串差异**：

```javascript
// 两种 H.265 codec 字符串
const codecs = [
  'video/mp4; codecs="hvc1.1.6.L93.B0"',  // H.265 Main Profile
  'video/mp4; codecs="hev1.1.6.L93.B0"',  // H.265 Main Profile (另一种标识)
];

codecs.forEach(codec => {
  const supported = MediaSource.isTypeSupported(codec);
  console.log(`${codec}: ${supported ? '✅' : '❌'}`);
});

// Chrome/Firefox 输出：
// video/mp4; codecs="hvc1.1.6.L93.B0": ❌
// video/mp4; codecs="hev1.1.6.L93.B0": ❌

// Safari (macOS) 输出：
// video/mp4; codecs="hvc1.1.6.L93.B0": ✅
// video/mp4; codecs="hev1.1.6.L93.B0": ❌
```

**关键结论**：
- Chrome/Firefox/Edge：完全不支持 H.265 MSE 硬解，必须用 WASM 软解
- Safari (macOS/iOS)：支持 `hvc1` 格式的 H.265，但 `hev1` 不支持
- 实际项目中：服务端转码为 H.264 是最保险的方案

### 6.4 WASM 软解的性能分析

**CPU 占用对比（单路 1080p 30fps 流）**：

| 解码模式 | 编码格式 | CPU 占用 | 内存占用 | 电量消耗 | 是否发热 |
|---------|---------|---------|---------|---------|---------|
| **MSE 硬解** | H.264 | < 5% | ~50MB | 低 | 否 |
| **WASM 软解** | H.264 | 15-20% | ~80MB | 中 | 轻微 |
| **WASM 软解 (SIMD)** | H.265 | 20-30% | ~100MB | 中高 | 明显 |
| **WASM 软解 (无 SIMD)** | H.265 | 40-60% | ~120MB | 高 | 严重 |

**多路并发性能（16 路 1080p 30fps）**：

| 解码模式 | 编码格式 | 总 CPU 占用 | 是否流畅 | 适用场景 |
|---------|---------|------------|---------|---------|
| **MSE 硬解** | H.264 | < 30% | ✅ 流畅 | **推荐**：16 路监控 |
| **WASM 软解** | H.264 | 250-300% | ⚠️ 勉强 | 不推荐：CPU 占满 |
| **WASM 软解 (SIMD)** | H.265 | 350-450% | ❌ 卡顿 | **不可行**：16 路 H.265 |
| **混合模式** | 前 8 路 H.264 (MSE) + 后 8 路 H.265 (WASM) | 150-200% | ⚠️ 勉强 | 折中方案 |

```javascript
class SmartStreamManager {
  constructor() {
    this.players = [];
    this.mseLimit = this.detectMSELimit();  // 检测 MSE 并发上限
  }

  // 检测 MSE 并发上限（简化版）
  detectMSELimit() {
    const userAgent = navigator.userAgent;
    if (userAgent.includes('Chrome')) {
      return 8;  // Chrome 约 8 个
    } else if (userAgent.includes('Firefox')) {
      return 10;  // Firefox 约 10 个
    } else if (userAgent.includes('Safari')) {
      return 12;  // Safari 约 12 个
    }
    return 8;  // 默认保守值
  }

  addStream(streamID, container, url, codec) {
    const currentMSECount = this.players.filter(p => p.mode === 'mse').length;

    // 如果是 H.264 且未达到 MSE 上限，用 MSE 硬解
    if (codec === 'h264' && currentMSECount < this.mseLimit) {
      const player = this.createMSEPlayer(container, url);
      this.players.push({ streamID, player, mode: 'mse' });
      console.log(`流 ${streamID} 使用 MSE 硬解（${currentMSECount + 1}/${this.mseLimit}）`);
    } 
    // 否则用 jessibuca WASM 软解
    else {
      const player = this.createWASMPlayer(container, url);
      this.players.push({ streamID, player, mode: 'wasm' });
      console.log(`流 ${streamID} 使用 WASM 软解（H.265 或 MSE 已满）`);
    }
  }

  createMSEPlayer(container, url) {
    return mpegts.createPlayer({
      type: 'flv',
      isLive: true,
      url: url,
    }, {
      enableWorker: true,
      liveBufferLatencyChasing: true,
    });
  }

  createWASMPlayer(container, url) {
    return new Jessibuca({
      container: container,
      useMSE: false,   // 强制 WASM 软解
      useWCS: true,
      useSIMD: true,
      videoBuffer: 0.2,
    });
  }
}
```

```javascript
class AdaptivePlayer {
  constructor(container, url) {
    this.container = container;
    this.url = url;
    this.player = null;
    this.mode = 'mse';  // 默认 MSE 硬解
    
    this.init();
    this.startCPUMonitor();
  }

  init() {
    this.player = new Jessibuca({
      container: this.container,
      useMSE: true,   // 优先 MSE
      useWCS: true,   // 支持 WASM 回退
      useSIMD: true,
    });

    this.player.on('performance', (stats) => {
      this.mode = stats.decodeType;  // 'mse' 或 'wasm'
      this.cpuUsage = stats.cpuUsage || 0;
    });

    this.player.play(this.url);
  }

  startCPUMonitor() {
    setInterval(() => {
      // 如果 CPU 占用 > 80%，提示用户降低并发数或降低分辨率
      if (this.cpuUsage > 80) {
        console.warn(`CPU 占用过高: ${this.cpuUsage}%，当前模式: ${this.mode}`);
        this.showCPUWarning();
      }
    }, 5000);  // 每 5 秒检测一次
  }

  showCPUWarning() {
    // 提示用户
    const toast = document.createElement('div');
    toast.textContent = `CPU 占用 ${this.cpuUsage}%，建议减少分屏数或降低画质`;
    toast.style.cssText = 'position:fixed;top:20px;right:20px;background:#ff5722;color:#fff;padding:10px;border-radius:5px;z-index:9999;';
    document.body.appendChild(toast);
    setTimeout(() => toast.remove(), 3000);
  }
}
```

### 6.6 生产环境建议

**方案 1：服务端统一转码为 H.264（推荐）**

```bash

  -b:v 2000k -maxrate 2500k -bufsize 4000k \
  -g 30 -keyint_min 30 -sc_threshold 0 \
  -c:a copy \
  -f flv rtmp://server/live/stream
```

- 前端无需 WASM 软解，CPU 占用低
- 16 路 H.264 流无压力，MSE 硬解流畅
- 兼容所有浏览器（Chrome/Firefox/Safari/Edge）

- 服务端 CPU 占用增加（转码开销）
- 带宽增加（H.264 体积比 H.265 大 30-50%）

## 八、requestAnimationFrame 渲染循环

### 8.1 问题：直接在 onmessage 里画会掉帧

```javascript
// ❌ 错误：每次收到消息立刻画
ws.onmessage = (e) => {
  const boxes = JSON.parse(e.data).boxes;
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  boxes.forEach(box => drawBox(ctx, box, canvas.width, canvas.height));
};
// 问题：消息频率不规律（10ms 一次或 100ms 一次），画面抖动
```

### 8.2 正确：解耦消息接收和渲染

```javascript
let latestBoxes = [];

// WebSocket 只更新数据
ws.onmessage = (e) => {
  latestBoxes = JSON.parse(e.data).boxes;
};

// 独立的渲染循环（60fps 稳定）
function render() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);

  latestBoxes.forEach(box => {
    drawBox(ctx, box, canvas.width, canvas.height);
  });

  requestAnimationFrame(render);
}

// 启动渲染循环
render();
```

## 九、16 路分屏的 Canvas 管理

### 9.1 CanvasOverlayManager 类

```javascript
class CanvasOverlayManager {
  constructor() {
    this.overlays = new Map();  // streamID → { canvas, ctx, latestBoxes, animationId }
  }

  addOverlay(streamID, videoElement) {
    // 创建 Canvas
    const canvas = document.createElement('canvas');
    const ctx = canvas.getContext('2d');

    // 定位覆盖在 video 上
    canvas.style.position = 'absolute';
    canvas.style.pointerEvents = 'none';
    canvas.style.zIndex = '10';
    videoElement.parentElement.appendChild(canvas);

    // 尺寸同步函数
    const syncSize = () => {
      canvas.width = videoElement.clientWidth;
      canvas.height = videoElement.clientHeight;
      canvas.style.width = videoElement.clientWidth + 'px';
      canvas.style.height = videoElement.clientHeight + 'px';
      canvas.style.top = videoElement.offsetTop + 'px';
      canvas.style.left = videoElement.offsetLeft + 'px';
    };

    videoElement.addEventListener('loadedmetadata', syncSize);
    window.addEventListener('resize', syncSize);
    syncSize();

    // 渲染循环
    let latestBoxes = [];
    const render = () => {
      if (!canvas.isConnected) return;  // Canvas 已从 DOM 移除，停止渲染

      ctx.clearRect(0, 0, canvas.width, canvas.height);
      latestBoxes.forEach(box => {
        this.drawBox(ctx, box, canvas.width, canvas.height);
      });

      const animationId = requestAnimationFrame(render);
      overlay.animationId = animationId;
    };

    const overlay = { canvas, ctx, latestBoxes, animationId: null };
    this.overlays.set(streamID, overlay);
    render();
  }

  updateBoxes(streamID, boxes) {
    const overlay = this.overlays.get(streamID);
    if (overlay) {
      overlay.latestBoxes = boxes;  // 只更新数据，不触发绘制
    }
  }

  removeOverlay(streamID) {
    const overlay = this.overlays.get(streamID);
    if (overlay) {
      if (overlay.animationId) {
        cancelAnimationFrame(overlay.animationId);
      }
      overlay.canvas.remove();
      this.overlays.delete(streamID);
    }
  }

  drawBox(ctx, box, canvasWidth, canvasHeight) {
    const x = box.x * canvasWidth;
    const y = box.y * canvasHeight;
    const w = box.w * canvasWidth;
    const h = box.h * canvasHeight;

    // 边框
    ctx.strokeStyle = box.confidence > 0.9 ? '#ff0000' : '#ffaa00';
    ctx.lineWidth = 2;
    ctx.strokeRect(x, y, w, h);

    // 标签背景
    const label = `${box.label} ${(box.confidence * 100).toFixed(0)}%`;
    ctx.font = '14px Arial';
    const textMetrics = ctx.measureText(label);
    const textWidth = textMetrics.width;

    ctx.fillStyle = ctx.strokeStyle;
    ctx.fillRect(x, y - 22, textWidth + 8, 20);

    // 标签文字
    ctx.fillStyle = '#ffffff';
    ctx.fillText(label, x + 4, y - 6);
  }
}

// 使用示例
const overlayManager = new CanvasOverlayManager();

// 为 16 路流添加 Canvas
for (let i = 1; i <= 16; i++) {
  const video = document.getElementById(`video-${i}`);
  overlayManager.addOverlay(`cam${i}`, video);
}

// WebSocket 收到检测结果
ws.onmessage = (e) => {
  const result = JSON.parse(e.data);
  overlayManager.updateBoxes(result.stream_id, result.boxes);
};
```

## 十一、多目标追踪轨迹绘制

### 11.1 TrackHistory 类

```javascript
class TrackHistory {
  constructor(maxLength = 30) {
    this.tracks = new Map();  // trackID → [{ x, y, timestamp }, ...]
    this.maxLength = maxLength;
  }

  update(trackID, x, y) {
    if (!this.tracks.has(trackID)) {
      this.tracks.set(trackID, []);
    }
    const history = this.tracks.get(trackID);
    history.push({ x, y, timestamp: Date.now() });

    // 保留最近 N 个点
    if (history.length > this.maxLength) {
      history.shift();
    }
  }

  draw(ctx, canvasWidth, canvasHeight) {
    ctx.strokeStyle = '#00ff00';
    ctx.lineWidth = 2;

    for (const history of this.tracks.values()) {
      if (history.length < 2) continue;

      ctx.beginPath();
      ctx.moveTo(history[0].x * canvasWidth, history[0].y * canvasHeight);

      for (let i = 1; i < history.length; i++) {
        ctx.lineTo(history[i].x * canvasWidth, history[i].y * canvasHeight);
      }

      ctx.stroke();
    }
  }

  cleanup(activeTrackIDs) {
    // 清理已消失的目标轨迹
    for (const trackID of this.tracks.keys()) {
      if (!activeTrackIDs.includes(trackID)) {
        this.tracks.delete(trackID);
      }
    }
  }
}

// 在渲染循环中使用
const trackHistory = new TrackHistory();

function render() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);

  // 先画轨迹
  trackHistory.draw(ctx, canvas.width, canvas.height);

  // 再画检测框
  latestBoxes.forEach(box => {
    drawBox(ctx, box, canvas.width, canvas.height);
    if (box.track_id) {
      trackHistory.update(box.track_id, box.x + box.w / 2, box.y + box.h / 2);
    }
  });

  trackHistory.cleanup(latestBoxes.map(b => b.track_id).filter(Boolean));

  requestAnimationFrame(render);
}
```

### 核心要点

| 主题 | 关键点 |
|------|--------|
| **MSE API** | MediaSource/SourceBuffer 状态机，FLV 必须转 fMP4 才能 append |

| 指数退避 | Exponential Backoff | 重连间隔指数增长（1s → 2s → 4s → 8s） |
| 归一化坐标 | Normalized Coordinates | 0-1 范围的相对坐标，与分辨率无关 |
| requestAnimationFrame | RAF | 浏览器提供的 60fps 渲染循环 API |
